cmake_minimum_required(VERSION 3.25)
include(cmake/Synth2Version.cmake)
set(SYNTH2_RELEASE_TAG "" CACHE STRING "Release tag for source archive/manual release builds; empty means a dated development build")
synth2_configure_version("${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/generated/version")
project(Synth2 VERSION ${SYNTH2_NUMERIC_VERSION} LANGUAGES C CXX)
# Out-of-source layout based on iPlug2/iPlug2OOS (see dependencies.lock.json).
set(IPLUG2_DIR "${CMAKE_CURRENT_SOURCE_DIR}/iPlug2" CACHE PATH "Path to the iPlug2 framework checkout")
# Experimental Linux native builds are selected through the linux presets.
if(WIN32 OR APPLE)
    set(synth2_plugin_default ON)
else()
    set(synth2_plugin_default OFF)
endif()
option(SYNTH2_BUILD_IPLUG2 "Build the iPlug2 desktop instrument" ${synth2_plugin_default})
option(SYNTH2_BUILD_TESTS "Build portable core regression tests" ON)
option(SYNTH2_BUILD_NATIVE_SMOKE "Test built plugins through their public state/MIDI/audio APIs" OFF)
option(SYNTH2_BUILD_TOOLS "Build offline development tools" ON)
if(SYNTH2_BUILD_IPLUG2 AND MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
include(cmake/Synth2Core.cmake)
if(SYNTH2_BUILD_TOOLS OR SYNTH2_BUILD_TESTS)
    include(cmake/Synth2Tools.cmake)
endif()
if(SYNTH2_BUILD_IPLUG2)
    add_subdirectory(projects/Synth2)
endif()
if(SYNTH2_BUILD_TESTS OR SYNTH2_BUILD_NATIVE_SMOKE)
    include(CTest)
endif()
if(SYNTH2_BUILD_TESTS)
    include(cmake/PortableTests.cmake)
endif()
if(SYNTH2_BUILD_NATIVE_SMOKE)
    include(cmake/NativeSmokeTests.cmake)
endif()
